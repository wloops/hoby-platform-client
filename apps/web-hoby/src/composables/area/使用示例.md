```ts
import {
  getAreaList,
  getAreaTree,
  getAreaNameByCode,
  getAreaCodeByName,
} from '@/utils/areaData';

// 1. 获取所有省份
const provinces = await getAreaList({ cityID: '全国' });
console.log('所有省份:', provinces);

// 2. 获取广东省下的城市
const cities = (await getAreaList({
  cityID: 'local.province',
  province: '广东省',
})) as City[]; // 使用类型断言

// 3. 获取广州市下的区县
const districts = (await getAreaList({
  cityID: 'local.city',
  city: '广州市',
})) as District[];

// 4. 获取树形结构
const areaTree = await getAreaTree();
areaTree.forEach((province) => {
  console.log(province.name);
  province.children.forEach((city) => {
    console.log('  ', city.name);
  });
});

// 5. 通过code获取名称
const name = await getAreaNameByCode('440106');
console.log('440106对应的名称:', name); // 输出: 天河区

// 6. 通过名称获取code
const code = await getAreaCodeByName('天河区', 'district');
console.log('天河区对应的code:', code); // 输出: 440106
```
